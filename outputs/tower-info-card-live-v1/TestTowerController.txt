using UnityEngine;
public sealed class TowerController : MonoBehaviour
{
    public TowerStats BaseStats;
    public TowerData Data;
    public int SpawnIndex=4;
    public bool CanChangeTargetPriority=true;
    private TargetPriority priority=TargetPriority.Closest;
    public TowerStats GetFinalStats()=>BaseStats;
    public TowerData GetTowerData()=>Data;
    public int GetRemainingAction()=>2;
    public int GetMaxAction()=>3;
    public TargetPriority GetTargetPriority()=>priority;
    public void SetTargetPriority(TargetPriority value){priority=value;}
}
